function runFukudaImportAutomated_() {
  try {
    // 1. ロック取得（同時実行防止）
    var lock = LockService.getScriptLock();
    lock.waitLock(60000);
    
    // 2. インポート実行
    importFukudaHistoricalData();
    
    // 3. 結果確認
    var result = checkImportResult_();
    // result: { status: "SUCCESS", companies_added: 100, warnings: 0 }
    
    // 4. 福田さんへメール
    sendImportCompletionEmail_(result);
    
    // 5. 小柳さんへ報告
    notifyKoyanagi_(result);
    
    // 6. ステータス更新
    updateStatusPhase7_(result);
    
  } catch (e) {
    // エラー処理
    sendErrorNotificationEmail_(e.toString());
    throw e;
  } finally {
    lock.releaseLock();
  }
}
