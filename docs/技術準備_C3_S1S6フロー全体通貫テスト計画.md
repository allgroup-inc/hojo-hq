```python
# kanri_system/interface/s1_input_adapter.py

"""
S1 からのデータ入力 adapter
期待される入力形式: JSON / CSV / Database query

入力例:
{
  "日付": "2026-10-17",
  "営業ID": "empl_001",
  "客先": "顧客A",
  "訪問/通話時間": 120,
  "案件": "新規提案",
  "進捗": "打ち合わせ完了",
  "次回予定": "2026-10-20"
}
"""

class S1InputAdapter:
    def __init__(self, data_source):
        # data_source: JSON / CSV / DB connection
        self.data_source = data_source
    
    def normalize(self, raw_data):
        """
        S1 形式 → S6 内部形式に変換
        """
        normalized = {
            'date': raw_data.get('日付'),
            'employee_id': raw_data.get('営業ID'),
            'customer_name': raw_data.get('客先'),
            'call_duration': int(raw_data.get('訪問/通話時間', 0)),
            'case_name': raw_data.get('案件'),
            'progress': raw_data.get('進捗'),
            'next_scheduled': raw_data.get('次回予定'),
        }
        return normalized
    
    def validate(self, normalized_data):
        """
        必須フィールド・データ型を検証
        """
        required = ['date', 'employee_id', 'customer_name']
        for field in required:
            if not normalized_data.get(field):
                raise ValueError(f"Missing required field: {field}")
        return True
```

### テスト 2: S2/S3/S4/S5 での変換・集約ロジック

```python
# kanri_system/pipeline/data_aggregation.py

"""
中間システム (S2/S3/S4/S5) での変換・集約
"""

class DataAggregator:
    def aggregate_call_logs(self, s1_entries):
        """
        S1 日報 → call_log 集約
        """
        aggregated = {}
        for entry in s1_entries:
            customer = entry['customer_name']
            if customer not in aggregated:
                aggregated[customer] = {
                    'total_duration': 0,
                    'last_contact': None,
                    'interactions': [],
                }
            aggregated[customer]['total_duration'] += entry['call_duration']
            aggregated[customer]['last_contact'] = entry['date']
            aggregated[customer]['interactions'].append({
                'date': entry['date'],
                'type': 'call',
                'duration': entry['call_duration'],
                'case': entry['case_name'],
                'progress': entry['progress'],
            })
        return aggregated

    def enrich_customer_data(self, aggregated_logs):
        """
        既存顧客マスタと照合・エンリッチメント
        """
        enriched = {}
        for customer_name, log_data in aggregated_logs.items():
            # Query customer master
            customer_record = self.query_customer_master(customer_name)
            if customer_record:
                enriched[customer_name] = {
                    **log_data,
                    'customer_id': customer_record['id'],
                    'status': customer_record['status'],
                    'contract_status': customer_record['contract_status'],
                }
            else:
                # 新規顧客フラグ
                enriched[customer_name] = {
                    **log_data,
                    'is_new': True,
                }
        return enriched
```

```python
Base = declarative_base()

    customer_id = Column(String(50), index=True)
    date = Column(DateTime, index=True)
    employee_id = Column(String(50))
    call_duration = Column(Integer)  # minutes
    case_name = Column(String(255))
    progress = Column(String(100))
    next_scheduled = Column(DateTime)
    created_at = Column(DateTime, default=datetime.utcnow)
    updated_at = Column(DateTime, default=datetime.utcnow, onupdate=datetime.utcnow)
```

```python
"""
S1→S6 同期の監査ログ
"""

class S1SyncAuditLog:
    def __init__(self):
        self.logs = []
    
    def log_success(self, batch_id, record_count):
        self.logs.append({
            'timestamp': datetime.utcnow(),
            'batch_id': batch_id,
            'status': 'SUCCESS',
            'record_count': record_count,
        })
    
    def log_error(self, batch_id, error_msg):
        self.logs.append({
            'timestamp': datetime.utcnow(),
            'batch_id': batch_id,
            'status': 'ERROR',
            'error_message': error_msg,
        })
    
    def log_partial_failure(self, batch_id, success_count, fail_count, failures):
        self.logs.append({
            'timestamp': datetime.utcnow(),
            'batch_id': batch_id,
            'status': 'PARTIAL_FAILURE',
            'success_count': success_count,
            'fail_count': fail_count,
            'failures': failures,  # List of failed record IDs + reasons
        })
```
